cmake_minimum_required(VERSION 3.25)
project(fogcityfurysrc CXX)
set(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")

function(add_fcf_xex target)
  add_executable(${target} "${REPO_ROOT}/src/game.cpp" "${CMAKE_CURRENT_LIST_DIR}/runtime_compat.cpp")
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_EXTENSIONS ON)
  target_compile_definitions(${target} PRIVATE FCF_GFX_XSDK ${ARGN})
  target_compile_options(${target} PRIVATE -Wall -Wno-unused-function -Wno-misleading-indentation)
  target_include_directories(${target} PRIVATE "${REPO_ROOT}/src" "${CMAKE_CURRENT_LIST_DIR}/build/gen")
endfunction()

add_fcf_xex(fogcityfury)
xsdk_add_xex(fogcityfury TITLE_ID 0x46434646 NAME "Fog City Fury" VERSION 1.3.6.0)

option(FCF_XSDK_OBJECT_TEST "Build a title cycling terrain/props/water/cars every three seconds" OFF)
if(FCF_XSDK_OBJECT_TEST)
  add_fcf_xex(fogcityfury_objects FCF_XSDK_OBJECT_TEST)
  xsdk_add_xex(fogcityfury_objects TITLE_ID 0x4643464F NAME "Fog City Fury GPU Test" VERSION 1.3.6.0)
endif()
